// CRUD operations of event table of college_event database /* 1. Initialize PROJECT > npm init -y 2. Install express and mysql8 > npm install express mysql8 EXECUTION: To open index.html GET http://localhost:3000/ */ const express = require("express"); const mysql = require("mysql8"); const app = express(); app.use(express.json()); // Read HTML files from public folder app.use(express.static("public")); // Connection establishment with college_event database const db = mysql.createConnection({ host: "localhost", user: "root", password: "root", database: "college_event" }); db.connect((err) => { if (err) throw err; console.log("Connected to MySQL"); }); // TO GET DETAILS OF ALL EVENTS app.get("/events", (req, res) => { db.query( "SELECT * FROM event", (err, result) => { if (err) { res.status(500).send(err.message); return; } res.json(result); } ); }); // TO GET THE DETAILS OF ONE EVENT app.get("/events/:event_id", (req, res) => { db.query( "SELECT * FROM event WHERE event_id=?", [req.params.event_id], (err, result) => { if (err) { res.status(500).send(err.message); return; } res.json(result); } ); }); // TO INSERT EVENT DATA INTO event TABLE app.post("/events", (req, res) => { const { event_id, event_name, event_type, event_date, venue, description } = req.body; db.query( `INSERT INTO event (event_id, event_name, event_type, event_date, venue, description) VALUES (?, ?, ?, ?, ?, ?)`, [ event_id, event_name, event_type, event_date, venue, description ], (err) => { if (err) { res.status(500).send(err.message); return; } res.send("Event Inserted"); } ); }); // TO UPDATE EVENT DETAILS app.put("/events/:event_id", (req, res) => { const { event_name, event_type, event_date, venue, description } = req.body; db.query( `UPDATE event SET event_name=?, event_type=?, event_date=?, venue=?, description=? WHERE event_id=?`, [ event_name, event_type, event_date, venue, description, req.params.event_id ], (err, result) => { if (err) { res.status(500).send(err.message); return; } if (result.affectedRows == 0) res.status(404).send("Event NOT found"); else res.send("Event Updated"); } ); }); // TO DELETE EVENT DETAILS app.delete("/events/:event_id", (req, res) => { db.query( "DELETE FROM event WHERE event_id=?", [req.params.event_id], (err, result) => { if (err) { res.status(500).send(err.message); return; } if (result.affectedRows == 0) res.status(404).send("Event NOT found to delete"); else res.send("Event Deleted"); } ); }); // Server to listen on port number 3000 app.listen(3000, () => { console.log("Server running at http://localhost:3000"); });
